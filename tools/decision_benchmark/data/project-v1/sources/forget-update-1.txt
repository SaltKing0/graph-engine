+def _suppressed_summary_members(nodes: dict[str, Node], edges: list[Edge]) -> set[str]:
+    """Forget suppresses the recorded source lineage, even after repartitioning.
+
+    Any overlap blocks a new abstraction until the operator explicitly restores
+    the forgotten summary. Legacy summaries lack derived_from; their saved
+    consolidator links retain at least the linked members, regardless of review
+    or validity state. Never infer suppression from ordinary forgotten notes.
+    """
+    forgotten = {n.id for n in nodes.values()
+                 if n.status == "tombstone" and SUMMARY_TAG in n.tags}
+    members = {member for nid in forgotten for member in nodes[nid].derived_from}
+    for edge in edges:
+        if (edge.source in forgotten and edge.origin == "consolidator"
+                and edge.kind == "extends"):
+            members.add(edge.target)
+    return members
